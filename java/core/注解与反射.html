Java 注解与反射 ​

注解（Annotation）和反射（Reflection）是 Java 元编程的核心技术。本文将详细讲解注解的定义与使用、反射的 API 以及它们在实际项目中的应用。

注解基础 ​

注解概述 ​

注解是附加在代码上的元数据，用于提供额外的信息。它可以被编译器、框架或运行时工具读取和处理。

java
// 注解的使用示例
@Deprecated                    // 标记过时的方法
@Override                      // 标记重写的方法
@SuppressWarnings("unchecked") // 抑制编译警告

public class AnnotationDemo {
    
    @Deprecated
    public void oldMethod() {
        // 这个方法已过时
    }
    
    @Override
    public String toString() {
        return "AnnotationDemo";
    }
    
    @SuppressWarnings("unchecked")
    public void suppressWarning() {
        List list = new ArrayList();
        list.add("No warning");
    }
}

标准注解 ​

注解作用目标
@Override标记方法重写父类方法方法
@Deprecated标记过时的 API类、方法、字段等
@SuppressWarnings抑制编译警告类、方法、语句
@SafeVarargs安全处理可变参数构造器、方法
@FunctionalInterface标记函数式接口接口

自定义注解 ​

定义注解 ​

java
// 使用 @interface 关键字定义注解

元注解 ​

元注解是用于修饰注解的注解：

java
import java.lang.annotation.*;

@Target({ElementType.TYPE, ElementType.METHOD, ElementType.FIELD})
// 指定注解可以应用的目标：
// TYPE: 类、接口、枚举
// METHOD: 方法
// FIELD: 字段
// CONSTRUCTOR: 构造方法
// PARAMETER: 参数
// LOCAL_VARIABLE: 局部变量
// ANNOTATION_TYPE: 注解类型
// PACKAGE: 包

@Retention(RetentionPolicy.RUNTIME)
// 指定注解的保留策略：
// SOURCE: 源码中存在，编译后丢弃
// BINARY: 编译后的 .class 文件中存在，但运行时不可用
// RUNTIME: 运行时存在，可通过反射读取

@Documented
// 生成 JavaDoc 时包含此注解

@Inherited
// 子类自动继承父类的注解

public @interface CustomAnnotation {
    String value() default "";
}

使用自定义注解 ​

java
// 定义注解
@Target(ElementType.TYPE)
@Retention(RetentionPolicy.RUNTIME)
public @interface Table {
    String name();
    String schema() default "public";
}

@Target(ElementType.FIELD)
@Retention(RetentionPolicy.RUNTIME)
public @interface Column {
    String name();
    boolean primaryKey() default false;
    boolean nullable() default true;
}

// 使用注解
@Table(name = "users", schema = "app")
public class User {
    
    @Column(name = "user_id", primaryKey = true, nullable = false)
    private Long id;
    
    @Column(name = "user_name", nullable = false)
    private String name;
    
    @Column(name = "user_email")
    private String email;
}

反射基础 ​

反射概述 ​

反射是在运行时检查和修改类、方法、字段等信息的能力。它使得 Java 具有动态性，可以在运行时创建对象、调用方法、访问字段。

java
// 反射能做什么
// 1. 获取类的信息（方法、字段、构造器等）
// 2. 动态创建对象
// 3. 动态调用方法
// 4. 动态访问和修改字段
// 5. 读取运行时注解

Class 类 ​

java
// 获取 Class 对象的三种方式

public class ReflectionDemo {
    
    public static void main(String[] args) throws Exception {
        // 方式一：Object.getClass()
        String str = "Hello";
        Class<?> clazz1 = str.getClass();
        
        // 方式二：类字面常量
        Class<?> clazz2 = String.class;
        
        // 方式三：Class.forName()
        Class<?> clazz3 = Class.forName("java.lang.String");
        
        // 获取类的基本信息
        Class<?> clazz = User.class;
        
        System.out.println("类名: " + clazz.getName());         // com.example.User
        System.out.println("简单名: " + clazz.getSimpleName()); // User
        System.out.println("包名: " + clazz.getPackage().getName());
        System.out.println("父类: " + clazz.getSuperclass().getName());
        System.out.println("接口: " + Arrays.toString(clazz.getInterfaces()));
        System.out.println("修饰符: " + Modifier.toString(clazz.getModifiers()));
        
        // 判断类型
        System.out.println("是否是接口: " + clazz.isInterface());
        System.out.println("是否是数组: " + clazz.isArray());
        System.out.println("是否是枚举: " + clazz.isEnum());
    }
}

获取构造器 ​

java
public class ConstructorReflection {
    
    public static void main(String[] args) throws Exception {
        Class<?> clazz = User.class;
        
        // 获取所有公有构造器
        Constructor<?>[] constructors = clazz.getConstructors();
        
        // 获取所有声明的构造器（包括私有）
        Constructor<?>[] declaredConstructors = clazz.getDeclaredConstructors();

        Constructor<User> privateConstructor = clazz.getDeclaredConstructor();
        privateConstructor.setAccessible(true);  // 突破访问限制
        User defaultUser = privateConstructor.newInstance();
    }
}

获取字段 ​

java
public class FieldReflection {
    
    public static void main(String[] args) throws Exception {
        User user = new User("张三", 25);
        Class<?> clazz = user.getClass();
        
        // 获取字段信息
        Field[] fields = clazz.getDeclaredFields();
        for (Field field : fields) {
            System.out.printf("字段: %s, 类型: %s, 修饰符: %s%n",
                field.getName(),
                field.getType().getSimpleName(),
                Modifier.toString(field.getModifiers())
            );
        }
        
        // 读取字段值
        Field nameField = clazz.getDeclaredField("name");
        nameField.setAccessible(true);  // 突破 private 限制
        String name = (String) nameField.get(user);
        System.out.println("姓名: " + name);
        
        // 修改字段值
        nameField.set(user, "李四");
        System.out.println("修改后: " + user.getName());
        
        // 静态字段
        Field staticField = clazz.getDeclaredField("MAX_COUNT");
        staticField.setAccessible(true);
        int maxCount = (int) staticField.get(null);  // 静态字段传 null
    }
}

获取方法 ​

java
public class MethodReflection {
    
    public static void main(String[] args) throws Exception {
        User user = new User("张三", 25);
        Class<?> clazz = user.getClass();
        
        // 获取所有方法（包括继承的）
        Method[] methods = clazz.getMethods();
        
        // 获取声明的方法（不包括继承的）
        Method[] declaredMethods = clazz.getDeclaredMethods();
        
        // 获取指定方法
        Method getNameMethod = clazz.getMethod("getName");
        Method setNameMethod = clazz.getMethod("setName", String.class);
        
        // 调用方法
        String name = (String) getNameMethod.invoke(user);
        System.out.println("当前姓名: " + name);
        
        setNameMethod.invoke(user, "李四");
        System.out.println("修改后: " + user.getName());
        
        // 调用私有方法
        Method privateMethod = clazz.getDeclaredMethod("doSomething");
        privateMethod.setAccessible(true);
        privateMethod.invoke(user);
        
        // 调用静态方法
        Method staticMethod = clazz.getDeclaredMethod("getInstance");
        Object instance = staticMethod.invoke(null);  // 静态方法传 null
    }
}

        boolean hasAnnotation = clazz.isAnnotationPresent(Table.class);
        
        // 获取所有注解
        Annotation[] annotations = clazz.getAnnotations();
    }
}

ORM 框架示例 ​

java
/**
 * 简化版 ORM 框架：根据注解自动生成 SQL
 */
public class SimpleORM {
    
    // 根据注解生成 SELECT 语句
    public static String generateSelect(Class<?> entityClass) {
        // 读取表名
        Table table = entityClass.getAnnotation(Table.class);
        if (table == null) {
            throw new IllegalArgumentException("类上没有 @Table 注解");
        }
        String tableName = table.name();
        
        // 读取列名
        Field[] fields = entityClass.getDeclaredFields();
        List<String> columns = new ArrayList<>();
        List<String> conditions = new ArrayList<>();
        
        for (Field field : fields) {
            Column column = field.getAnnotation(Column.class);
            if (column != null) {
                columns.add(column.name());
                if (column.primaryKey()) {
                    conditions.add(column.name() + " = ?");
                }
            }
        }
        
        // 生成 SQL
        return String.format("SELECT %s FROM %s WHERE %s",
            String.join(", ", columns),
            tableName,
            String.join(" AND ", conditions)
        );
    }
    
    // 根据注解生成 INSERT 语句
    public static String generateInsert(Object entity) {
        Class<?> clazz = entity.getClass();
        Table table = clazz.getAnnotation(Table.class);
        if (table == null) {
            throw new IllegalArgumentException("类上没有 @Table 注解");
        }
        
        List<String> columns = new ArrayList<>();
        List<String> values = new ArrayList<>();
        
        for (Field field : clazz.getDeclaredFields()) {
            Column column = field.getAnnotation(Column.class);
            if (column != null && !column.primaryKey()) {
                columns.add(column.name());
                values.add("?");
            }
        }
        
        return String.format("INSERT INTO %s (%s) VALUES (%s)",
            table.name(),
            String.join(", ", columns),
            String.join(", ", values)
        );
    }
    
    // 从 ResultSet 映射对象
    public static <T> T mapFromResultSet(ResultSet rs, Class<T> clazz) throws Exception {
        T instance = clazz.getDeclaredConstructor().newInstance();
        
        for (Field field : clazz.getDeclaredFields()) {
            Column column = field.getAnnotation(Column.class);
            if (column != null) {
                Object value = rs.getObject(column.name());
                field.setAccessible(true);
                field.set(instance, value);
            }
        }
        
        return instance;
    }
}

依赖注入容器示例 ​

java
/**
 * 简化版依赖注入容器
 */
public class SimpleContainer {
    
    private final Map<Class<?>, Object> singletons = new ConcurrentHashMap<>();
    
    // 注册单例
    public <T> void register(Class<T> clazz) {
        singletons.put(clazz, createInstance(clazz));
    }
    
    // 获取实例
    @SuppressWarnings("unchecked")
    public <T> T get(Class<T> clazz) {
        return (T) singletons.computeIfAbsent(clazz, this::createInstance);
    }
    
    // 创建实例并注入依赖
    private Object createInstance(Class<?> clazz) {
        try {
            // 查找带 @Inject 注解的构造器
            Constructor<?> constructor = findInjectableConstructor(clazz);
            constructor.setAccessible(true);
            
            // 注入构造器参数
            Object[] args = resolveParameters(constructor);
            Object instance = constructor.newInstance(args);
            
            // 注入字段
            injectFields(instance);
            
            // 注入方法参数
            injectMethods(instance);
            
            return instance;
        } catch (Exception e) {
            throw new RuntimeException("创建实例失败: " + clazz.getName(), e);
        }
    }
    
    private Constructor<?> findInjectableConstructor(Class<?> clazz) throws Exception {
        // 查找 @Inject 构造器
        for (Constructor<?> ctor : clazz.getDeclaredConstructors()) {
            if (ctor.isAnnotationPresent(Inject.class)) {
                return ctor;
            }
        }
        // 如果没有，使用无参构造器
        return clazz.getDeclaredConstructor();
    }
    
    private Object[] resolveParameters(Constructor<?> constructor) {
        return Arrays.stream(constructor.getParameterTypes())
                     .map(this::get)
                     .toArray();
    }
    
    private void injectFields(Object instance) throws Exception {
        for (Field field : instance.getClass().getDeclaredFields()) {
            if (field.isAnnotationPresent(Inject.class)) {
                field.setAccessible(true);
                field.set(instance, get(field.getType()));
            }
        }
    }
    
    private void injectMethods(Object instance) throws Exception {
        for (Method method : instance.getClass().getDeclaredMethods()) {
            if (method.isAnnotationPresent(Inject.class)) {
                method.setAccessible(true);
                Object[] args = resolveParameters(method);
                method.invoke(instance, args);
            }
        }
    }
}

// 自定义 @Inject 注解
@Target({ElementType.CONSTRUCTOR, ElementType.FIELD, ElementType.METHOD})
@Retention(RetentionPolicy.RUNTIME)
public @interface Inject {}

反射的性能 ​

性能对比 ​

java
public class ReflectionPerformance {
    
    public static void main(String[] args) throws Exception {
        User user = new User("张三", 25);
        Method getter = User.class.getMethod("getName");
        
        int iterations = 10_000_000;
        
        // 直接调用
        long start1 = System.nanoTime();
        for (int i = 0; i < iterations; i++) {
            user.getName();
        }
        long directTime = System.nanoTime() - start1;
        
        // 反射调用
        long start2 = System.nanoTime();
        for (int i = 0; i < iterations; i++) {
            getter.invoke(user);
        }
        long reflectTime = System.nanoTime() - start2;
        
        // 反射（去除访问检查）
        getter.setAccessible(true);
        long start3 = System.nanoTime();
        for (int i = 0; i < iterations; i++) {
            getter.invoke(user);
        }
        long accessibleTime = System.nanoTime() - start3;
        
        System.out.printf("直接调用: %d ms%n", directTime / 1_000_000);
        System.out.printf("反射调用: %d ms (%.1fx 慢)%n", 
            reflectTime / 1_000_000, (double) reflectTime / directTime);
        System.out.printf("反射(可访问): %d ms (%.1fx 慢)%n", 
            accessibleTime / 1_000_000, (double) accessibleTime / directTime);
    }
}

性能优化建议 ​

  1. 缓存反射对象：避免重复获取 Class、Method、Field
  2. 使用 setAccessible(true)：减少访问检查开销
  3. 使用 MethodHandle：Java 7+ 的现代 API，性能接近直接调用
  4. 避免在热路径中使用反射：对性能要求高的场景考虑代码生成
java
// MethodHandle 示例
public class MethodHandleExample {
    
    public static void main(String[] args) throws Throwable {
        User user = new User("张三", 25);
        MethodHandles.Lookup lookup = MethodHandles.lookup();
        
        // 获取方法句柄
        MethodHandle getter = lookup.findVirtual(
            User.class, "getName", MethodType.methodType(String.class)
        );
        
        // 调用
        String name = (String) getter.invoke(user);
        
        // 性能测试
        int iterations = 10_000_000;
        long start = System.nanoTime();
        for (int i = 0; i < iterations; i++) {
            getter.invoke(user);
        }
        long elapsed = System.nanoTime() - start;
        System.out.printf("MethodHandle: %d ms%n", elapsed / 1_000_000);
    }
}

public class ClassCache {
    private static final Map<Class<?>, ClassInfo> cache = new ConcurrentHashMap<>();
    
    public static ClassInfo getInfo(Class<?> clazz) {
        return cache.computeIfAbsent(clazz, c -> {
            ClassInfo info = new ClassInfo();
            info.fields = c.getDeclaredFields();
            info.methods = c.getDeclaredMethods();
            info.constructors = c.getDeclaredConstructors();
            return info;
        });
    }
    
    private static class ClassInfo {
        Field[] fields;
        Method[] methods;
        Constructor<?>[] constructors;
    }
}
  1. 使用 try-with-resources 管理：
java
public class SafeReflection {
    
    public static <T> Optional<T> getAnnotation(AnnotatedElement element, Class<T> annotationType) {
        T annotation = element.getAnnotation(annotationType);
        return Optional.ofNullable(annotation);
    }
    
    public static void invokeMethod(Object target, String methodName, Object... args) {
        try {
            Method method = target.getClass().getMethod(methodName, getParameterTypes(args));
            method.invoke(target, args);
        } catch (NoSuchMethodException e) {
            // 方法不存在，记录日志但不中断
            log.warn("方法不存在: {}.{}", target.getClass().getName(), methodName);
        } catch (Exception e) {
            throw new RuntimeException("调用方法失败", e);
        }
    }
    
    private static Class<?>[] getParameterTypes(Object... args) {
        return Arrays.stream(args)
                     .map(Object::getClass)
                     .toArray(Class<?>[]::new);
    }
}
