AQS 概述 ​

AQS（AbstractQueuedSynchronizer）是 Java 并发包的核心基础，ReentrantLock、CountDownLatch、Semaphore 等都基于它实现。

AQS 核心结构 ​

java
public abstract class AbstractQueuedSynchronizer {
    
    // state 字段：表示同步状态
    private volatile int state;
    
    // CLH 队列头结点
    private transient volatile Node head;
    
    // CLH 队列尾结点
    private transient volatile Node tail;
    
    // 内部类：Node 表示队列中的节点
    static final class Node {
        // 节点状态
        static final int CANCELLED =  1;  // 已取消
        static final int SIGNAL    = -1;  // 后继节点需要被唤醒
        static final int CONDITION = -2;  // 等待条件
        static final int PROPAGATE = -3;  // 释放共享状态
        
        volatile int waitStatus;        // 等待状态
        volatile Node prev;             // 前驱节点
        volatile Node next;             // 后继节点
        volatile Thread thread;         // 等待的线程
        Node nextWaiter;                // 条件队列中的下一个节点
    }
    
    // CAS 操作：设置 state
    protected final boolean compareAndSetState(int expect, int update) {
        return compareAndSwapInt(stateOffset, expect, update);
    }
}

ReentrantLock 实现原理 ​

java
public class ReentrantLockDemo {
    
    // ReentrantLock 内部使用 AQS 实现
    private final Sync sync;
    
    // Sync 是 ReentrantLock 的内部类，继承自 AQS
    abstract static class Sync extends AbstractQueuedSynchronizer {
        
        // 尝试获取锁（需要子类实现）
        abstract boolean nonfairTryAcquire(int acquires);
        
        // 释放锁
        protected final boolean tryRelease(int releases) {
            int c = getState() - releases;
            if (Thread.currentThread() != getExclusiveOwnerThread())
                throw new IllegalMonitorStateException();
            boolean free = false;
            if (c == 0) {
                free = true;
                setExclusiveOwnerThread(null);
            }
            setState(c);
            return free;
        }
    }
    
    // 非公平锁实现
    static final class NonfairSync extends Sync {
        protected final boolean tryAcquire(int acquires) {
            // 直接 CAS 抢占
            if (compareAndSetState(0, acquires)) {
                setExclusiveOwnerThread(Thread.currentThread());
                return true;
            }
            return false;
        }
    }
    
    // 公平锁实现
    static final class FairSync extends Sync {
        protected final boolean tryAcquire(int acquires) {
            Thread current = Thread.currentThread();
            int c = getState();
            if (c == 0) {
                // 必须前驱节点为自己才获取
                if (!hasQueuedPredecessors() &&
                    compareAndSetState(0, acquires)) {
                    setExclusiveOwnerThread(current);
                    return true;
                }
            } else if (current == getExclusiveOwnerThread()) {
                int nextc = c + acquires;
                setState(nextc);
                return true;
            }
            return false;
        }
    }
}

Unsafe 类 ​

public class UnsafeExample {
    
    // Unsafe 是原子类的基础，提供 CAS 操作
    @SuppressWarnings("restriction")
    private static final Unsafe UNSAFE;
    private static final long VALUE_OFFSET;
    
    static {
        try {
            Field unsafeField = Unsafe.class.getDeclaredField("theUnsafe");
            unsafeField.setAccessible(true);
            UNSAFE = (Unsafe) unsafeField.get(null);
            VALUE_OFFSET = UNSAFE.objectFieldOffset(AtomicInteger.class.getDeclaredField("value"));
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }
    
    // CAS 操作：比较并交换
    public boolean compareAndSetInt(Object obj, long offset, int expected, int update) {
        return UNSAFE.compareAndSwapInt(obj, offset, expected, update);
    }
    
    // CAS 原理：硬件级别的原子操作
    // CPU 指令：CMPXCHG
    // 如果内存值 == expected，则设置为 update
}

原子类详解 ​

java
import java.util.concurrent.atomic.*;

public class AtomicClassesExample {
    
    public static void main(String[] args) {
        // AtomicInteger：原子整数
        AtomicInteger atomicInt = new AtomicInteger(0);
        
        atomicInt.get();              // 获取值
        atomicInt.set(10);            // 设置值
        atomicInt.getAndSet(20);      // 获取旧值并设置新值
        atomicInt.compareAndSet(20, 30);  // CAS 操作
        
        atomicInt.incrementAndGet();  // 原子自增并返回新值
        atomicInt.getAndIncrement();  // 原子自增并返回旧值
        atomicInt.decrementAndGet();  // 原子自减并返回新值
        atomicInt.getAndAdd(5);       // 原子加并返回旧值
        
        // AtomicLong：原子长整数
        AtomicLong atomicLong = new AtomicLong(0);
        // 方法与 AtomicInteger 类似
        
        // AtomicBoolean：原子布尔值
        AtomicBoolean atomicBool = new AtomicBoolean(false);
        atomicBool.compareAndSet(false, true);
        
        // AtomicReference：原子引用
        AtomicReference<String> atomicRef = new AtomicReference<>("Hello");
        atomicRef.compareAndSet("Hello", "World");
        
        // AtomicStampedReference：带时间戳的原子引用
        // 解决 ABA 问题
        AtomicStampedReference<String> stampedRef = 
            new AtomicStampedReference<>("Hello", 0);
        stampedRef.compareAndSet("Hello", "World", 0, 1);
        
        // AtomicIntegerArray：原子整数数组
        AtomicIntegerArray intArray = new AtomicIntegerArray(10);
        intArray.getAndSet(0, 100);
        
        // AtomicIntegerFieldUpdater：原子更新器
        AtomicIntegerFieldUpdater<Counter> updater = 
            AtomicIntegerFieldUpdater.newUpdater(Counter.class, "count");
        Counter counter = new Counter();
        updater.incrementAndGet(counter);
    }
    
    static class Counter {
        volatile int count;
    }
}

ABA 问题与解决方案 ​

java
public class ABAProblem {
    
    // ABA 问题：
    // 1. 线程1 读取值 A
    // 2. 线程2 将 A→B→A
    // 3. 线程1 CAS(A→C) 成功
    // 4. 但实际值已经改变过了
    
    // 使用 AtomicStampedReference 解决
    public void solveABA() {
        AtomicStampedReference<String> ref = 
            new AtomicStampedReference<>("A", 0);
        
        // 增加版本号
        int stamp = ref.getStamp();
        String reference = ref.getReference();
        
        // 同时检查引用值和版本号
        ref.compareAndSet(reference, "C", stamp, stamp + 1);
    }
    
    // AtomicMarkableReference 方案
    // 使用标记位代替版本号
    public void solveABAWithMark() {
        AtomicMarkableReference<String> ref = 
            new AtomicMarkableReference<>("A", false);
        
        boolean mark = ref.isMarked();
        ref.compareAndSet("A", "C", mark, !mark);
    }
}

ConcurrentHashMap 原理 ​

java
public class ConcurrentHashMapDeep {
    
    // JDK 8+ 实现：CAS + synchronized
    // 结构：Node[] table + 链表 + 红黑树
    
    public static void main(String[] args) {
        ConcurrentHashMap<String, Integer> map = new ConcurrentHashMap<>();
        
        // 核心方法
        map.put("key", 1);
        map.get("key");
        map.remove("key");
        
        // CAS 操作
        map.putIfAbsent("key2", 2);     // 不存在时添加
        map.replace("key", 1, 10);       // 条件替换
        map.remove("key", 1);            // 条件删除
        
        // 原子更新
        map.compute("counter", (k, v) -> v == null ? 1 : v + 1);
        map.computeIfAbsent("counter2", k -> 100);
        map.computeIfPresent("counter", (k, v) -> v * 2);
        
        // 并行遍历
        map.forEach(1, (k, v) -> System.out.println(k + ": " + v));
        map.search(1, (k, v) -> v > 50 ? k : null);
        map.reduce(1, (k, v) -> v, Integer::sum);
    }
    
    // ConcurrentHashMap 工作流程：
    // 1. put(key, value)
    //    - 计算 hash 值
    //    - 如果 table 未初始化，CAS 初始化
    //    - 根据 hash 找到位置
    //    - 如果该位置为空，CAS 写入
    //    - 如果该位置不为空，加 synchronized 锁
    //    - 遍历链表或红黑树
    //    - 如果找到 key，更新 value
    //    - 如果未找到，添加新节点
}

CopyOnWriteArrayList ​

java
import java.util.concurrent.CopyOnWriteArrayList;

public class CopyOnWriteArrayListExample {
    
    public static void main(String[] args) {
        // CopyOnWriteArrayList：写入时复制
        // 适用场景：读多写少
        CopyOnWriteArrayList<String> list = new CopyOnWriteArrayList<>();
        
        // 读操作：无锁
        list.add("A");
        list.add("B");
        list.add("C");
        
        // 写操作：加锁 + 复制
        list.add("D");  // 创建新数组，复制所有元素
        
        // 迭代时无需检查并发修改
        for (String item : list) {
            System.out.println(item);
            list.add("X");  // 不会抛出 ConcurrentModificationException
        }
        
        // 适用场景：
        // 1. 白名单/黑名单
        // 2. 配置信息
        // 3. 事件监听器列表
    }
}

BlockingQueue ​

java
import java.util.concurrent.*;

public class BlockingQueueExample {
    
    public static void main(String[] args) throws Exception {
        // ArrayBlockingQueue：有界阻塞队列
        BlockingQueue<Integer> arrayQueue = new ArrayBlockingQueue<>(10);
        
        // LinkedBlockingQueue：可选有界
        BlockingQueue<Integer> linkedQueue = new LinkedBlockingQueue<>();
        
        // PriorityBlockingQueue：优先级阻塞队列
        BlockingQueue<Integer> priorityQueue = new PriorityBlockingQueue<>();
        
        // SynchronousQueue：同步队列（不存储元素）
        BlockingQueue<Integer> synchronousQueue = new SynchronousQueue<>();
        
        // TransferQueue：可传输元素的队列
        TransferQueue<Integer> transferQueue = new LinkedTransferQueue<>();
        
        // 基本操作
        arrayQueue.put(1);           // 添加，满时阻塞
        arrayQueue.offer(2, 1, TimeUnit.SECONDS);  // 超时添加
        int val = arrayQueue.take(); // 获取，空时阻塞
        int val2 = arrayQueue.poll(1, TimeUnit.SECONDS);  // 超时获取
        
        // 生产者-消费者示例
        BlockingQueue<Integer> buffer = new ArrayBlockingQueue<>(10);
        
        // 生产者
        Thread producer = new Thread(() -> {
            for (int i = 0; i < 20; i++) {
                try {
                    buffer.put(i);
                    System.out.println("生产者: " + i);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            }
        });
        
        // 消费者
        Thread consumer = new Thread(() -> {
            while (true) {
                try {
                    Integer val3 = buffer.take();
                    System.out.println("消费者: " + val3);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    break;
                }
            }
        });
        
        producer.start();
        consumer.start();
        
        producer.join();
        Thread.sleep(1000);
        consumer.interrupt();
    }
}

// 不可变对象天生线程安全
public final class ImmutablePoint {
    private final int x;
    private final int y;
    
    public ImmutablePoint(int x, int y) {
        this.x = x;
        this.y = y;
    }
    
    // 只提供 getter，无 setter
    public int getX() { return x; }
    public int getY() { return y; }
    
    // 无修改方法
    // 所有方法都返回新对象
    public ImmutablePoint move(int dx, int dy) {
        return new ImmutablePoint(x + dx, y + dy);
    }
}

线程安全的单例 ​

java
public class ThreadSafeSingleton {
    
    // 方式一：饿汉式（立即加载）
    public class EagerSingleton {
        private static final EagerSingleton INSTANCE = new EagerSingleton();
        
        private EagerSingleton() {}
        
        public static EagerSingleton getInstance() {
            return INSTANCE;
        }
    }
    
    // 方式二：懒汉式 + synchronized
    public class LazySingleton {
        private static LazySingleton instance;
        
        private LazySingleton() {}
        
        public static synchronized LazySingleton getInstance() {
            if (instance == null) {
                instance = new LazySingleton();
            }
            return instance;
        }
    }
    
    // 方式三：双重检查锁（推荐）
    public class DoubleCheckedLockingSingleton {
        private static volatile DoubleCheckedLockingSingleton instance;
        
        private DoubleCheckedLockingSingleton() {}
        
        public static DoubleCheckedLockingSingleton getInstance() {
            if (instance == null) {  // 第一次检查
                synchronized (DoubleCheckedLockingSingleton.class) {
                    if (instance == null) {  // 第二次检查
                        instance = new DoubleCheckedLockingSingleton();
                    }
                }
            }
            return instance;
        }
    }
    
    // 方式四：静态内部类（推荐）
    public class InnerClassSingleton {
        private InnerClassSingleton() {}
        
        private static class Holder {
            private static final InnerClassSingleton INSTANCE = new InnerClassSingleton();
        }
        
        public static InnerClassSingleton getInstance() {
            return Holder.INSTANCE;
        }
    }
    
    // 方式五：枚举（最安全）
    public enum EnumSingleton {
        INSTANCE;
        
        private final String value;
        
        EnumSingleton() {
            this.value = "Singleton";
        }
        
        public String getValue() {
            return value;
        }
    }
}

LongAdder vs AtomicLong ​

java
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.atomic.LongAdder;

public class LongAdderExample {
    
    public static void main(String[] args) {
        // AtomicLong：单变量 CAS
        // 问题：高并发下 CAS 失败率高
        AtomicLong atomicCounter = new AtomicLong(0);
        
        // LongAdder：分段累加
        // 原理：将 value 分散到多个单元格，减少竞争
        LongAdder adderCounter = new LongAdder();
        
        // 使用
        atomicCounter.incrementAndGet();
        adderCounter.increment();
        
        // 获取最终值
        long atomicResult = atomicCounter.get();
        long adderResult = adderCounter.sum();
        
        // 性能对比：
        // 线程数越多，LongAdder 优势越明显
        // 16 线程下，LongAdder 比 AtomicLong 快 5-10 倍
    }
    
    // LongAdder 内部结构：
    // ┌─────────────────────────────────────────┐
    // │  base (普通 long)                        │
    // │  + Cells[] (单元格数组)                  │
    // │    ├── Cell0: long value                 │
    // │    ├── Cell1: long value                 │
    // │    ├── Cell2: long value                 │
    // │    └── ...                              │
    // │ 最终结果 = base + sum(cells)             │
    // └─────────────────────────────────────────┘
}

CompletableFuture 进阶 ​

java
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

public class CompletableFutureAdvanced {
    
    public static void main(String[] args) throws Exception {
        // 异步编排示例
        CompletableFuture<String> getUserFuture = CompletableFuture.supplyAsync(() -> {
            // 获取用户信息
            return fetchUser();
        });
        
        CompletableFuture<String> getOrdersFuture = CompletableFuture.supplyAsync(() -> {
            // 获取订单信息
            return fetchOrders();
        });
        
        CompletableFuture<String> getRecommendationsFuture = CompletableFuture.supplyAsync(() -> {
            // 获取推荐信息
            return fetchRecommendations();
        });
        
        // 并行执行，全部完成
        CompletableFuture.allOf(getUserFuture, getOrdersFuture, getRecommendationsFuture)
            .thenRun(() -> {
                try {
                    String user = getUserFuture.get();
                    String orders = getOrdersFuture.get();
                    String recommendations = getRecommendationsFuture.get();
                    
                    // 合并结果
                    String result = String.format("%s\n%s\n%s", user, orders, recommendations);
                    System.out.println(result);
                } catch (Exception e) {
                    e.printStackTrace();
                }
            })
            .exceptionally(ex -> {
                System.err.println("执行失败: " + ex.getMessage());
                return null;
            })
            .join();
    }
    
    private static String fetchUser() {
        try { Thread.sleep(500); } catch (InterruptedException e) {}
        return "用户: 张三";
    }
    
    private static String fetchOrders() {
        try { Thread.sleep(800); } catch (InterruptedException e) {}
        return "订单: 3 笔";
    }
    
    private static String fetchRecommendations() {
        try { Thread.sleep(300); } catch (InterruptedException e) {}
        return "推荐: A, B, C";
    }
}
