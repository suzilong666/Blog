一、参数绑定机制 ​

1.1 参数绑定概述 ​

SpringMVC的参数绑定（Data Binding） 是指将HTTP请求参数自动转换为控制器方法参数的过程。

绑定流程 ​

HTTP请求参数
    │
    ├── 查询字符串 (query string)
    ├── 表单参数 (form data)
    ├── 路径变量 (path variable)
    ├── 请求体 (request body)
    └── 请求头 (request header)
    │
    ▼
SpringMVC参数绑定
    │
    ├── 1. 参数定位
    │   ├── @RequestParam
    │   ├── @PathVariable
    │   ├── @RequestBody
    │   └── 其他注解
    │
    ├── 2. 类型转换
    │   ├── String → Integer
    │   ├── String → Date
    │   └── String → 自定义类型
    │
    ├── 3. 数据格式化
    │   ├── @DateTimeFormat
    │   ├── @NumberFormat
    │   └── 自定义Formatter
    │
    └── 4. 参数校验
        ├── @NotNull
        ├── @Size
        └── 自定义Validator

1.2 基本类型绑定 ​

java
@RestController
@RequestMapping("/api/basic")
public class BasicBindingController {
    
    // 绑定基本类型
    @GetMapping("/types")
    public String basicTypes(
            @RequestParam String name,
            @RequestParam int age,
            @RequestParam double score,
            @RequestParam boolean active,
            @RequestParam long count) {
        
        return String.format("name=%s, age=%d, score=%.1f, active=%b, count=%d",
            name, age, score, active, count);
    }
    
    // 绑定数组
    @GetMapping("/array")
    public String arrayBinding(
            @RequestParam String[] tags,
            @RequestParam List<Integer> ids) {
        
        return String.format("tags=%s, ids=%s", 
            Arrays.toString(tags), ids);
    }
    
    // 绑定枚举
    @GetMapping("/enum")
    public String enumBinding(
            @RequestParam OrderStatus status) {
        
        return "status=" + status.name();
    }
}

// 1. 创建自定义转换器
public class StringToLocalDateConverter 
        implements Converter<String, LocalDate> {
    
    private final DateTimeFormatter formatter;
    
    public StringToLocalDateConverter(String pattern) {
        this.formatter = DateTimeFormatter.ofPattern(pattern);
    }
    
    @Override
    public LocalDate convert(String source) {
        if (source == null || source.isEmpty()) {
            return null;
        }
        return LocalDate.parse(source, formatter);
    }
}

实现Formatter接口（带Locale） ​

java
public class PhoneNumberFormatter 
        implements Formatter<PhoneNumber> {
    
    @Override
    public PhoneNumber parse(String text, Locale locale) {
        // 解析 "138-0000-0000" 或 "13800000000"
        String digits = text.replaceAll("[^\\d]", "");
        if (digits.length() == 11) {
            return new PhoneNumber(
                digits.substring(0, 3),
                digits.substring(3, 7),
                digits.substring(7)
            );
        }
        throw new IllegalArgumentException("Invalid phone number: " + text);
    }
    
    @Override
    public String print(PhoneNumber phone, Locale locale) {
        return String.format("%s-%s-%s", 
            phone.getPrefix(), 
            phone.getMiddle(), 
            phone.getSuffix());
    }
}

// 注册Formatter
@Configuration
public class WebConfig implements WebMvcConfigurer {
    
    @Override
    public void addFormatters(FormatterRegistry registry) {
        registry.addFormatter(new PhoneNumberFormatter());
    }
}

@Configuration
public class WebConfig implements WebMvcConfigurer {
    
    @Bean
    public Formatter<LocalDate> localDateFormatter() {
        return new Formatter<LocalDate>() {
            @Override
            public LocalDate parse(String text, Locale locale) {
                return LocalDate.parse(text, 
                    DateTimeFormatter.ofPattern("yyyy-MM-dd"));
            }
            
            @Override
            public String print(LocalDate date, Locale locale) {
                return date.format(
                    DateTimeFormatter.ofPattern("yyyy-MM-dd"));
            }
        };
    }
    
    @Bean
    public Formatter<LocalDateTime> localDateTimeFormatter() {
        return new Formatter<LocalDateTime>() {
            private final DateTimeFormatter formatter = 
                DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss");
            
            @Override
            public LocalDateTime parse(String text, Locale locale) {
                return LocalDateTime.parse(text, formatter);
            }
            
            @Override
            public String print(LocalDateTime date, Locale locale) {
                return date.format(formatter);
            }
        };
    }
    
    @Override
    public void addFormatters(FormatterRegistry registry) {
        registry.addFormatter(localDateFormatter());
        registry.addFormatter(localDateTimeFormatter());
    }
}

// 1. 在DTO上添加校验注解
@Data
public class UserCreateRequest {

    private String username;
    
    @NotBlank(message = "密码不能为空")
    @Size(min = 6, max = 20, message = "密码长度6-20位")
    private String password;
    
    @Email(message = "邮箱格式不正确")
    private String email;
    
    @Min(value = 0, message = "年龄不能为负")
    @Max(value = 150, message = "年龄不能超过150")
    private Integer age;
    
    @Valid  // 级联校验
    private Address address;
    
    @Size(max = 5, message = "最多选择5个兴趣")
    private List<String> hobbies;
}

@Data
public class Address {
    
    @NotBlank(message = "省份不能为空")
    private String province;
    
    @NotBlank(message = "城市不能为空")
    private String city;
    
    @Size(max = 200, message = "地址不能超过200个字符")
    private String detail;
}

// 2. 在Controller中使用@Valid触发校验
@RestController
@RequestMapping("/api/users")
public class UserController {
    
    @PostMapping
    @ResponseStatus(HttpStatus.CREATED)
    public User createUser(
            @Valid @RequestBody UserCreateRequest request,
            BindingResult bindingResult) {
        
        // 3. 处理校验错误
        if (bindingResult.hasErrors()) {
            List<String> errors = bindingResult.getFieldErrors()
                .stream()
                .map(error -> error.getField() + ": " + error.getDefaultMessage())
                .toList();
            throw new ValidationException(errors);
        }
        
        User user = userService.create(request);
        return user;
    }
}

// 1. 创建自定义校验注解
@Target({ElementType.FIELD, ElementType.PARAMETER})
@Retention(RetentionPolicy.RUNTIME)
@Constraint(validatedBy = PhoneNumberValidator.class)
public @interface PhoneNumber {
    String message() default "手机号格式不正确";
    Class<?>[] groups() default {};
    Class<? extends Payload>[] payload() default {};
}

// 2. 实现校验器
public class PhoneNumberValidator 
        implements ConstraintValidator<PhoneNumber, String> {
    
    private static final Pattern PHONE_PATTERN = 
        Pattern.compile("^1[3-9]\\d{9}$");
    
    @Override
    public void initialize(PhoneNumber constraintAnnotation) {
        // 初始化逻辑（可选）
    }
    
    @Override
    public boolean isValid(String phone, 
            ConstraintValidatorContext context) {
        if (phone == null || phone.isEmpty()) {
            return true;  // 空值由@NotNull等处理
        }
        return PHONE_PATTERN.matcher(phone).matches();
    }
}

// 3. 使用自定义注解
@Data

4.5 分组校验 ​

// 1. 定义校验分组
public interface CreateGroup {}  // 创建时校验
public interface UpdateGroup {}  // 更新时校验
public interface LoginGroup {}  // 登录时校验

// 2. 在DTO上使用分组
@Data
public class UserDTO {
    
    @NotNull(groups = UpdateGroup.class, message = "更新时ID必填")
    private Long id;
    
    @NotBlank(groups = {CreateGroup.class, UpdateGroup.class})
    private String username;
    
    @NotBlank(groups = CreateGroup.class, message = "创建时密码必填")
    @Size(min = 6, groups = CreateGroup.class)
    private String password;
    
    @Email(groups = {CreateGroup.class, UpdateGroup.class})
    private String email;
    
    @NotBlank(groups = LoginGroup.class, message = "登录时用户名必填")
    private String loginUsername;
}

// 3. 在Controller中指定分组
@RestController
public class UserController {
    
    // 创建用户：使用CreateGroup分组
    @PostMapping("/users")
    public User createUser(
            @Validated(CreateGroup.class) @RequestBody UserDTO dto) {
        return userService.create(dto);
    }
    
    // 更新用户：使用UpdateGroup分组
    @PutMapping("/users/{id}")
    public User updateUser(
            @PathVariable Long id,
            @Validated(UpdateGroup.class) @RequestBody UserDTO dto) {
        return userService.update(id, dto);
    }
    
    // 登录：使用LoginGroup分组
    @PostMapping("/login")
    public TokenResponse login(
            @Validated(LoginGroup.class) @RequestBody UserDTO dto) {
        return authService.login(dto);
    }
}

五、文件上传 ​

5.1 单个文件上传 ​

java
@RestController
@RequestMapping("/api/files")
public class FileUploadController {
    
    // 配置上传目录
    @Value("${upload.path:/uploads}")
    private String uploadPath;
    
    // 单个文件上传
    @PostMapping("/single")
    public UploadResult uploadFile(
            @RequestParam("file") MultipartFile file,
            @RequestParam(value = "description", required = false) String description) {
        
        // 1. 验证文件
        if (file.isEmpty()) {
            throw new RuntimeException("文件不能为空");
        }
        
        // 2. 验证文件类型
        String contentType = file.getContentType();
        if (!contentType.startsWith("image/")) {
            throw new RuntimeException("只支持图片格式");
        }
        
        // 3. 验证文件大小（在配置中限制更优）
        long maxSize = 5 * 1024 * 1024;  // 5MB
        if (file.getSize() > maxSize) {
            throw new RuntimeException("文件不能超过5MB");
        }
        
        // 4. 保存文件
        String originalFilename = file.getOriginalFilename();
        String extension = originalFilename.substring(
            originalFilename.lastIndexOf("."));
        String newFilename = UUID.randomUUID().toString() + extension;
        
        Path targetPath = Path.of(uploadPath, newFilename);
        Files.createDirectories(targetPath.getParent());
        Files.copy(file.getInputStream(), targetPath);
        
        // 5. 返回结果
        return UploadResult.builder()
            .originalFilename(originalFilename)
            .newFilename(newFilename)
            .size(file.getSize())
            .contentType(contentType)
            .url("/api/files/download/" + newFilename)
            .description(description)
            .uploadTime(LocalDateTime.now())
            .build();
    }
}

5.2 多文件上传 ​

java
// 多文件上传
@PostMapping("/multiple")
public List<UploadResult> uploadMultipleFiles(
        @RequestParam("files") MultipartFile[] files,
        @RequestParam(value = "directory", defaultValue = "default") String directory) {
    
    List<UploadResult> results = new ArrayList<>();
    
    for (MultipartFile file : files) {
        if (!file.isEmpty()) {
            String originalFilename = file.getOriginalFilename();
            String extension = originalFilename.substring(
                originalFilename.lastIndexOf("."));
            String newFilename = UUID.randomUUID().toString() + extension;
            
            Path targetPath = Path.of(uploadPath, directory, newFilename);
            Files.createDirectories(targetPath.getParent());
            Files.copy(file.getInputStream(), targetPath);
            
            results.add(UploadResult.builder()
                .originalFilename(originalFilename)
                .newFilename(newFilename)
                .size(file.getSize())
                .contentType(file.getContentType())
                .url("/api/files/download/" + directory + "/" + newFilename)
                .build());
        }
    }
    
    return results;
}

六、参数绑定原理 ​

6.1 参数解析流程 ​

java
public class RequestMappingHandlerAdapter {
    
    private List<HandlerMethodArgumentResolver> argumentResolvers;
    
    @Override
    protected ModelAndView handleInternal(...) {
        // 解析方法参数
        Object[] args = getMethodArgumentValues(
            request, response, handlerMethod, webRequest, mav);
        
        // 执行方法
        Object returnValue = handlerMethod.getMethod()
            .invoke(handlerMethod.getBean(), args);
    }
    
    private Object[] getMethodArgumentValues(
            HttpServletRequest request,
            HttpServletResponse response,
            HandlerMethod handlerMethod,
            WebRequest webRequest,
            ModelAndView mav) throws Exception {
        
        MethodParameter[] parameters = handlerMethod.getMethodParameters();
        Object[] args = new Object[parameters.length];
        
        for (int i = 0; i < parameters.length; i++) {
            MethodParameter parameter = parameters[i];
            
            // 查找支持该参数的Resolver
            HandlerMethodArgumentResolver resolver = 
                findArgumentResolver(parameter);
            
            // 使用Resolver解析参数
            args[i] = resolver.resolveArgument(
                parameter, mav, webRequest, dataBinderFactory);
        }
        
        return args;
    }
}

6.3 自定义参数解析器 ​

java
// 1. 创建自定义参数注解
@Target({ElementType.PARAMETER, ElementType.TYPE})
@Retention(RetentionPolicy.RUNTIME)
public @interface CurrentUser {
}

// 2. 实现自定义参数解析器
public class CurrentUserArgumentResolver 
        implements HandlerMethodArgumentResolver {
    
    @Autowired
    private UserService userService;
    
    @Override
    public boolean supportsParameter(MethodParameter parameter) {
        // 支持@CurrentUser注解的User类型参数
        return parameter.hasParameterAnnotation(CurrentUser.class) &&
               parameter.getParameterType().equals(User.class);
    }
    
    @Override
    public Object resolveArgument(
            MethodParameter parameter,
            ModelAndViewContainer mavContainer,
            NativeWebRequest webRequest,
            WebDataBinderFactory binderFactory) 
            throws Exception {
        
        // 从SecurityContext获取当前用户ID
        Long userId = SecurityContextHolder.getCurrentUserId();
        
        // 查询用户信息
        return userService.findById(userId);
    }
}

// 3. 注册解析器
@Configuration
public class WebConfig implements WebMvcConfigurer {
    
    @Autowired
    private CurrentUserArgumentResolver currentUserResolver;
    
    @Override
    public void addArgumentResolvers(
            List<HandlerMethodArgumentResolver> resolvers) {
        resolvers.add(currentUserResolver);
    }
}

// 4. 使用自定义参数
@RestController
@RequestMapping("/api/profile")
public class ProfileController {
    
    @GetMapping
    public User getCurrentUser(@CurrentUser User currentUser) {
        // 直接获取当前登录用户
        return currentUser;
    }
    
    @PutMapping
    public User updateProfile(
            @CurrentUser User currentUser,
            @RequestBody UpdateProfileRequest request) {
        return userService.update(currentUser.getId(), request);
    }
}

七、小结 ​

SpringMVC的参数绑定与数据校验是构建健壮Web应用的关键：

  1. 参数绑定：支持基本类型、POJO、数组、列表等多种类型的自动绑定
  2. 类型转换：内置大量转换器，支持自定义Converter和Formatter
  3. 数据格式化：支持日期、数字等的格式化注解
  4. 数据校验：基于Bean Validation规范，支持分组校验和自定义校验
  5. 文件上传：支持单文件、多文件上传及类型校验
  6. 扩展机制：支持自定义参数解析器、类型转换器

掌握这些特性，能够帮助我们：

  • 简化控制器代码
  • 保证数据的合法性
  • 构建安全、可靠的API接口
